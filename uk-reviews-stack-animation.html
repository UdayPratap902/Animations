GSAP 3D ScrollTrigger Card Deck Architecture & Implementation Guide

Use this complete reference, mathematical parameters, and production-ready snippets (React / Next.js / TypeScript / Vanilla JS) to drop this exact animation into any web application or AI prompt.

01. Core Animation Mechanics

The animation creates a tactile 3D card deck landing on top of previous reviews as the user scrolls. Rather than standard sliding or carousel cards, it uses real 3D spatial perspective (perspective: 1200px) with scroll scrubbing and section pinning.

1. Initial Card Positioning (t = 0)

  • Card 0: y: 0, rotateX: 0deg, scale: 1, zIndex: 1
  • Cards 1..N: y: +650px (mobile +480px), rotateX: 38deg (mobile 28deg), scale: 0.92, zIndex: i + 1

2. Scrub Step Timeline (i = 1..N)

  • Previous Cards: scale: 1 - depthDiff * 0.04, y: -depthDiff * 18px
  • Incoming Card: y: 0, rotateX: 0deg, scale: 1, opacity: 1 (duration: 1)

02. Exact Configuration Formulas

Parameter Desktop (≥ 768px) Mobile (< 768px) Formula / Purpose
Initial Y-Offset 650px 480px Keeps incoming card offscreen below stage
Initial Tilt Angle 38 deg 28 deg rotateX tilt in 3D space
Scroll Distance Per Card 900px 700px totalScroll = (N - 1) * distPerCard
Tiered Deck Offset -18px * depthDiff -12px * depthDiff Upward shift revealing deck top tabs
Tiered Deck Scale 1 - (0.04 * depthDiff) 1 - (0.04 * depthDiff) Subtle scale-down for background deck cards
Scrub Smoothness 1.2 1.2 GSAP scrub lag for smooth inertia

3. Cards 1..N start below viewport with y: 650 (mobile 480), rotateX: 38deg (mobile 28deg), scale: 0.92, with ascending zIndex (zIndex = i + 1).
4. On GSAP scrub timeline:
   - When card i ascends to y: 0 and rotateX: 0, all previous cards (prev < i) scale down to (1 - depthDiff * 0.04) and shift upward to (-depthDiff * 18px).
   - Higher z-index ensures incoming cards land ON TOP of older cards in the deck.
5. Provide a responsive, accessible fallback when prefers-reduced-motion is true.
6. Support ambient parallax gradient glow in the background."

import React, { useRef, useState, useEffect } from "react";
import { useGSAP } from "@gsap/react";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

if (typeof window !== "undefined") {
  gsap.registerPlugin(ScrollTrigger);
}

export interface Testimonial {
  id: string;
  name: string;
  role: string;
  text: string;
  initials: string;
  rating: number;
}

export const ReviewsStack: React.FC = () => {
  const containerRef = useRef<HTMLElement>(null);
  const [isReducedMotion, setIsReducedMotion] = useState(false);

  useEffect(() => {
    const checkMotion = () => {
      setIsReducedMotion(window.matchMedia("(prefers-reduced-motion: reduce)").matches);
    };
    checkMotion();
    window.addEventListener("resize", checkMotion);
    return () => window.removeEventListener("resize", checkMotion);
  }, []);

  useGSAP(
    () => {
      if (typeof window === "undefined" || isReducedMotion) return;

      const isMobile = window.innerWidth < 768;
      const cardEls = gsap.utils.toArray<HTMLElement>(".review-stack-card");
      const totalCards = cardEls.length;
      if (totalCards === 0) return;

      // 1. Set initial 3D transform state
      gsap.set(cardEls[0], {
        transformOrigin: "50% 0%",
        y: 0,
        rotateX: 0,
        scale: 1,
        opacity: 1,
        zIndex: 1,
      });

      const initialYOffset = isMobile ? 480 : 650;
      const initialTiltAngle = isMobile ? 28 : 38;

      for (let i = 1; i < totalCards; i++) {
        gsap.set(cardEls[i], {
          transformOrigin: "50% 0%",
          y: initialYOffset,
          rotateX: initialTiltAngle,
          scale: 0.92,
          opacity: 1,
          zIndex: i + 1,
        });
      }

      // 2. Setup pinned scrub timeline
      const scrollDistancePerCard = isMobile ? 700 : 900;
      const totalScrollDistance = (totalCards - 1) * scrollDistancePerCard;

      const tl = gsap.timeline({
        scrollTrigger: {
          trigger: containerRef.current,
          start: () => "top top",
          end: () => `+=${totalScrollDistance}`,
          pin: true,
          pinSpacing: true,
          scrub: 1.2,
          anticipatePin: 1,
          invalidateOnRefresh: true,
        },
      });

      // 3. Sequentially build stack steps
      for (let i = 1; i < totalCards; i++) {
        const incomingCard = cardEls[i];
        const stepLabel = `step-${i}`;

        for (let prev = 0; prev < i; prev++) {
          const depthDiff = i - prev;
          tl.to(
            cardEls[prev],
            {
              scale: 1 - depthDiff * 0.04,
              y: -depthDiff * (isMobile ? 12 : 18),
              ease: "none",
              duration: 1,
            },
            stepLabel
          );
        }

        tl.to(
          incomingCard,
          {
            y: 0,
            rotateX: 0,
            scale: 1,
            opacity: 1,
            ease: "none",
            duration: 1,
          },
          stepLabel
        );
      }
    },
    { scope: containerRef, dependencies: [isReducedMotion] }
  );

  return (
    <section
      ref={containerRef}
      className="relative min-h-screen flex flex-col justify-between overflow-hidden bg-gradient-to-b from-white via-indigo-50/20 to-white py-12 sm:py-16"
    >
      {/* Header */}
      <div className="relative max-w-3xl mx-auto px-4 text-center z-20">
        <h2 className="font-serif text-3xl sm:text-5xl font-medium tracking-tight text-slate-900">
          Trusted by engineering teams building modern web apps.
        </h2>
      </div>

      {/* 3D Stage */}
      <div
        className="relative flex-1 flex items-center justify-center my-6 px-4 w-full"
        style={{ perspective: "1200px", perspectiveOrigin: "50% 50%" }}
      >
        <div className="relative w-full max-w-[760px] min-h-[340px] flex items-center justify-center">
          {TESTIMONIALS.map((t, idx) => (
            <div key={t.id} className="review-stack-card absolute inset-0 flex items-center justify-center">
              <div className="w-full max-w-[720px] rounded-3xl border border-slate-200 bg-white p-8 sm:p-12 shadow-2xl flex flex-col justify-between">
                <div className="flex items-center justify-between mb-4">
                  <span className="text-xs font-bold uppercase tracking-wider text-indigo-600">Verified Showcase</span>
                  <span className="text-xs font-mono text-slate-400">0{idx + 1} / 0{TESTIMONIALS.length}</span>
                </div>
                <p className="font-serif italic text-lg sm:text-xl text-slate-800 mb-6">"{t.text}"</p>
                <div className="flex items-center justify-between pt-4 border-t border-slate-100">
                  <div className="flex items-center gap-3">
                    <div className="w-10 h-10 rounded-full bg-indigo-50 text-indigo-600 font-bold flex items-center justify-center">{t.initials}</div>
                    <div>
                      <h4 className="font-bold text-sm text-slate-900">{t.name}</h4>
                      <p className="text-xs text-slate-500">{t.role}</p>
                    </div>
                  </div>
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>
    </section>
  );
};
export default ReviewsStack;

document.addEventListener("DOMContentLoaded", () => {
  gsap.registerPlugin(ScrollTrigger);

  const isMobile = window.innerWidth < 768;
  const container = document.getElementById("reviews-stack");
  const cardEls = gsap.utils.toArray(".review-stack-card");
  const totalCards = cardEls.length;

  if (!container || totalCards === 0) return;

  // 1. Initial 3D Setup
  gsap.set(cardEls[0], {
    transformOrigin: "50% 0%",
    y: 0,
    rotateX: 0,
    scale: 1,
    opacity: 1,
    zIndex: 1,
    willChange: "transform",
  });

  const initialYOffset = isMobile ? 480 : 650;
  const initialTiltAngle = isMobile ? 28 : 38;

  for (let i = 1; i < totalCards; i++) {
    gsap.set(cardEls[i], {
      transformOrigin: "50% 0%",
      y: initialYOffset,
      rotateX: initialTiltAngle,
      scale: 0.92,
      opacity: 1,
      zIndex: i + 1,
      willChange: "transform",
    });
  }

  // 2. Timeline with Pinning & Scrub
  const scrollDistancePerCard = isMobile ? 700 : 900;

  const tl = gsap.timeline({

      trigger: container,
      start: () => "top top",
      end: () => `+=${totalScrollDistance}`,
      pin: true,
      pinSpacing: true,
      scrub: 1.2,
      anticipatePin: 1,
      invalidateOnRefresh: true,
    },
  });

  // 3. Sequential Card Landings
  for (let i = 1; i < totalCards; i++) {
    const incomingCard = cardEls[i];

    for (let prev = 0; prev < i; prev++) {

          scale: 1 - depthDiff * 0.04,

          ease: "none",

        y: 0,

        scale: 1,
        opacity: 1,
        ease: "none",
        duration: 1,
      },
      stepLabel

  ScrollTrigger.refresh();
});
