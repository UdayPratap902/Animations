📐 Mathematical Layout Engine

Standalone Panel Architecture & Safe Pinning

Each section acts as a full standalone panel with its own dedicated layout, translated across a continuous horizontal track with hardware acceleration.

01. Panel Width

100% Viewport Span

Each section spans exactly 100vw, ensuring complete visual focus without horizontal cramped multi-column splits:

width: 100vw (or 1/N of track)
02. Travel Distance

Scroll Distance

Total horizontal travel distance equals total track width minus one viewport width:

(N - 1) * 100vw = 300vw
03. Progress Metric

GPU scaleX

Progress bar transforms smoothly from 1/N to 1.0 via GPU composite without any DOM reflow:

scaleX: 0.25 → 1.0
⚠️

Critical Pinning Rule: No overflow-hidden on Pin Trigger

Applying overflow: hidden directly to the ScrollTrigger pin trigger container creates a new CSS stacking context that corrupts getBoundingClientRect() coordinate calculations for all downstream sections on the page. In this architecture, horizontal clipping is safely isolated to the child track wrapper, allowing GSAP pin measurements to remain mathematically pure.

Production React & Vanilla Code

Select your framework below to copy the fully typed, battle-tested implementation with clean ScrollTrigger lifecycle teardown. 

import React, { useRef } from "react";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";
import { useGSAP } from "@gsap/react";

if (typeof window !== "undefined") {
  gsap.registerPlugin(ScrollTrigger);
}

export const HorizontalShowcase = () => {
  const containerRef = useRef<HTMLElement>(null);
  const trackRef = useRef<HTMLDivElement>(null);
  const progressBarRef = useRef<HTMLDivElement>(null);
  const labelRef = useRef<HTMLSpanElement>(null);

  useGSAP(() => {
    if (window.innerWidth < 768) return;
    const container = containerRef.current;
    const track = trackRef.current;
    if (!container || !track) return;

    const getDist = () => track.scrollWidth - window.innerWidth;

    const tl = gsap.timeline({
      scrollTrigger: {
        trigger: container,
        start: "top top",
        end: () => `+=${getDist()}`,
        scrub: 1,
        pin: true,
        pinSpacing: true,
        anticipatePin: 1,
        invalidateOnRefresh: true,
      },
    });

    tl.fromTo(track, { x: 0 }, { x: () => -getDist(), ease: "none", force3D: true }, 0);

    if (progressBarRef.current) {
      tl.fromTo(progressBarRef.current, 
        { scaleX: 0.25, transformOrigin: "left center" }, 
        { scaleX: 1, ease: "none", force3D: true }, 0);
    }
  }, { scope: containerRef });

  return (
    <section ref={containerRef} className="relative bg-background flex flex-col">
      <div className="flex-1 overflow-hidden">
        <div ref={trackRef} className="flex h-full w-[400%]">
          {/* Standalone Panels 1 to 4 */}
        </div>
      </div>
    </section>
  );
};
// Standalone Vanilla JS + GSAP Implementation
gsap.registerPlugin(ScrollTrigger);

const container = document.querySelector("#showcase");
const track = document.querySelector("#horizontal-track");
const bar = document.querySelector("#progress-bar");
const label = document.querySelector("#panel-label");

function initHorizontalScroll() {
  if (window.innerWidth < 768) return;

  const getDist = () => track.scrollWidth - window.innerWidth;

  const tl = gsap.timeline({
    scrollTrigger: {
      trigger: container,
      start: "top top",
      end: () => `+=${getDist()}`,
      scrub: 1,
      pin: true,
      pinSpacing: true,
      anticipatePin: 1,
    }
  });

  tl.to(track, { x: () => -getDist(), ease: "none", force3D: true }, 0);
  tl.fromTo(bar, { scaleX: 0.25 }, { scaleX: 1, ease: "none" }, 0);

  tl.eventCallback("onUpdate", () => {
    const current = Math.min(4, Math.floor(tl.progress() * 3.999) + 1);
    label.textContent = String(current).padStart(2, "0");
  });
}

window.addEventListener("DOMContentLoaded", initHorizontalScroll);
