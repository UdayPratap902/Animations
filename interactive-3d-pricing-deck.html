02 • Deconstruct

How the deck is built

Four animation layers that compose to create the full hover-elevation experience.

Permanent Anchor Positions

Cards are permanently fixed in their left/center/right positions. No position swapping occurs on hover — only transforms are applied in-place. This keeps the layout stable while creating the illusion of spatial depth.

Starter (Left) rotate(-4deg) translateY(6px) scale(0.96)
Pro (Center) rotate(0deg) translateY(0px) scale(1.0)
Enterprise (Right) rotate(4deg) translateY(6px) scale(0.96) 
// Resting deck state
if (hoveredIdx === null) {
  if (i === 0) { // Left — Starter
    transform = "rotate(-4deg) translateY(6px) scale(0.96)";
    zIndex = 10;
  } else if (i === 1) { // Center — Pro
    transform = "rotate(0deg) translateY(0px) scale(1.0)";
    zIndex = 20;
  } else { // Right — Enterprise
    transform = "rotate(4deg) translateY(6px) scale(0.96)";
    zIndex = 10;
  }
}

Hover Elevation Mechanics

The hovered card straightens to 0° rotation, lifts 14px, scales to 1.025, and gains the highest z-index. Non-hovered cards simultaneously recede to a deeper scale and slightly lower y-position. Both happen in the same 300ms cubic ease.

Hovered Card
rotate(0deg) translateY(-14px) scale(1.025) — z: 35
Non-Hovered Side Card
rotate(±4deg) translateY(8px) scale(0.95) — z: 10
Non-Hovered Center
rotate(0deg) translateY(6px) scale(0.97) — z: 15 
// Hovered card: straighten + lift
if (isHovered) {

  zIndex = 35;
  shadowClass = isPro
    ? "shadow-[0_28px_65px_-12px_rgba(92,224,184,0.35)..."
    : "shadow-[0_28px_65px_-12px_rgba(0,0,0,0.18)...";
}

// Non-hovered: recede in place
else {
  if (i === 0)
    transform = "rotate(-4deg) translateY(8px) scale(0.95)";
  if (i === 1)
    transform = "rotate(0deg) translateY(6px) scale(0.97)";
  if (i === 2)
    transform = "rotate(4deg) translateY(8px) scale(0.95)";
}

Benefit List Expansion

Benefits are revealed via an AnimatePresence wrapper in Framer Motion (or a CSS max-height transition in Vanilla). Individual list items stagger in with 25ms delays between each, translating from y:5 to y:0 with opacity 0→1.

Wrapper Transition: max-height 280ms, opacity 220ms
Item Stagger: delay: index * 25ms, duration: 200ms
Easing: cubic-bezier(0.16, 1, 0.3, 1) 
// Framer Motion: AnimatePresence expand
<AnimatePresence>
  {isHovered && (
    <motion.div
      initial={{ opacity: 0, height: 0 }}
      animate={{ opacity: 1, height: "auto" }}
      exit={{ opacity: 0, height: 0 }}
      transition={{ duration: 0.28, ease: [0.16, 1, 0.3, 1] }}
    >
      {features.map((feat, fIdx) => (
        <motion.li
          key={fIdx}
          initial={{ opacity: 0, y: 5 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.2, delay: fIdx * 0.025 }}
        />
      ))}
    </motion.div>
  )}
</AnimatePresence>

Elastic Magnetic CTAs

CTA buttons use GSAP quickTo with elastic ease (elastic.out(1, 0.3)) to track cursor position within the button bounding box. On mouse-leave, springs back to origin. Disabled on touch/coarse-pointer devices.

Ease: elastic.out(1, 0.3) — natural spring recoil
Duration: 0.9s — lag felt, not instant
Strength: 15px — ratio of cursor delta to button size 
// GsapMagnetic component core
const xTo = gsap.quickTo(el, "x", {
  duration: 0.9,
  ease: "elastic.out(1, 0.3)"
});
const yTo = gsap.quickTo(el, "y", {
  duration: 0.9,
  ease: "elastic.out(1, 0.3)"
});

el.addEventListener("mousemove", (e) => {
  const { height, width, left, top } = el.getBoundingClientRect();
  const x = e.clientX - (left + width / 2);
  const y = e.clientY - (top + height / 2);
  xTo((x / width) * strength);  // strength = 15
  yTo((y / height) * strength);
});
el.addEventListener("mouseleave", () => {
  xTo(0); yTo(0);  // spring back to origin
});

PricingDeck.tsx — React / Next.js / Framer Motion 
"use client";

import { useState, useSyncExternalStore } from "react";
import { motion, AnimatePresence } from "framer-motion";

// 1. Reduced-motion hook via useSyncExternalStore
function subscribeReducedMotion(callback) {
  const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
  mq.addEventListener("change", callback);
  return () => mq.removeEventListener("change", callback);
}

// 2. Generic demo plans data
const PLANS = [
  { id: "starter", name: "Starter", price: 49,
    features: ["Up to 3 active spaces", "5 GB cloud storage", "Community support"],
    highlighted: false },
  { id: "pro", name: "Pro", price: 99, badge: "Most Popular",
    features: ["Everything in Starter", "Unlimited active spaces", "60 FPS acceleration"],
    highlighted: true },
  { id: "enterprise", name: "Enterprise", price: "Custom", badge: "Enterprise",
    features: ["Dedicated cluster", "Custom shader pipelines", "SSO & audit logs"],
    highlighted: false },
];

export default function PricingDeck() {
  const isReducedMotion = useSyncExternalStore(
    subscribeReducedMotion,
    () => window.matchMedia("(prefers-reduced-motion: reduce)").matches,
    () => false
  );
  const [hoveredIdx, setHoveredIdx] = useState<number | null>(null);

  return (
    <section id="pricing" className="relative py-20 bg-[#F4FBF8] overflow-hidden">
      <div className="max-w-7xl mx-auto px-6">
        {!isReducedMotion ? (
          <div className="hidden lg:flex items-end justify-center min-h-[580px]">
            <div className="relative flex items-end justify-center -space-x-12 xl:-space-x-14">
              {PLANS.map((plan, i) => {
                const isPro = plan.id === "pro";
                const isHovered = hoveredIdx === i;

                let transform = hoveredIdx === null
                  ? ["rotate(-4deg) translateY(6px) scale(0.96)",
                     "rotate(0deg) translateY(0px) scale(1.0)",
                     "rotate(4deg) translateY(6px) scale(0.96)"][i]
                  : isHovered
                    ? "rotate(0deg) translateY(-14px) scale(1.025)"
                    : ["rotate(-4deg) translateY(8px) scale(0.95)",
                       "rotate(0deg) translateY(6px) scale(0.97)",
                       "rotate(4deg) translateY(8px) scale(0.95)"][i];

                return (
                  <div
                    key={plan.id}
                    onMouseEnter={() => setHoveredIdx(i)}
                    onMouseLeave={() => setHoveredIdx(null)}
                    style={{ transform, zIndex: isHovered ? 35 : isPro ? 20 : 10,
                             transformOrigin: "bottom center" }}
                    className="relative w-[360px] rounded-3xl bg-white p-6 flex flex-col
                              transition-all duration-300 ease-out cursor-pointer will-change-transform"
                  >
                    <h3 className="text-2xl font-bold">{plan.name}</h3>

                    {/* Benefit list reveal */}
                    <AnimatePresence>
                      {isHovered && (
                        <motion.div
                          initial={{ opacity: 0, height: 0 }}
                          animate={{ opacity: 1, height: "auto" }}
                          exit={{ opacity: 0, height: 0 }}
                          transition={{ duration: 0.28, ease: [0.16, 1, 0.3, 1] }}
                        >
                          {plan.features.map((feat, fIdx) => (
                            <motion.li
                              key={fIdx}
                              initial={{ opacity: 0, y: 5 }}
                              animate={{ opacity: 1, y: 0 }}
                              transition={{ duration: 0.2, delay: fIdx * 0.025 }}
                            >{feat}</motion.li>
                          ))}
                        </motion.div>
                      )}
                    </AnimatePresence>
                  </div>
                );
              })}
            </div>
          </div>
        ) : null}
      </div>
    </section>
  );
}
pricing-deck.js — Vanilla JS Zero Dependencies 
// Vanilla JS 3D Pricing Deck — zero framework dependencies
// Requires: cards with class="deck-card", benefits wrapper class="benefits-wrapper"

const RESTING = [
  "rotate(-4deg) translateY(6px) scale(0.96)",  // Starter
  "rotate(0deg) translateY(0px) scale(1.0)",    // Pro
  "rotate(4deg) translateY(6px) scale(0.96)",   // Enterprise
];
const RECEDE = [
  "rotate(-4deg) translateY(8px) scale(0.95)",
  "rotate(0deg) translateY(6px) scale(0.97)",
  "rotate(4deg) translateY(8px) scale(0.95)",
];
const HOVER_LIFT = "rotate(0deg) translateY(-14px) scale(1.025)";
const Z_RESTING = [10, 20, 10];
const Z_HOVER   = 35;
const Z_RECEDE  = [10, 15, 10];

function initPricingDeck() {
  const cards = document.querySelectorAll(".deck-card");
  if (!cards.length) return;

  // Skip all motion if prefers-reduced-motion
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;

  function applyResting() {
    cards.forEach((card, i) => {
      card.style.transform = RESTING[i];
      card.style.zIndex = Z_RESTING[i];
      const bw = card.querySelector(".benefits-wrapper");
      if (bw) { bw.style.maxHeight = "0"; bw.style.opacity = "0"; }
      card.querySelectorAll(".benefit-item")
          .forEach(el => el.classList.remove("revealed"));
    });
  }

  function applyHover(activeIdx) {
    cards.forEach((card, i) => {
      if (i === activeIdx) {
        card.style.transform = HOVER_LIFT;
        card.style.zIndex = Z_HOVER;
        const bw = card.querySelector(".benefits-wrapper");
        if (bw) { bw.style.maxHeight = "300px"; bw.style.opacity = "1"; }
        card.querySelectorAll(".benefit-item").forEach((el, fIdx) => {
          setTimeout(() => el.classList.add("revealed"), fIdx * 25);
        });
      } else {
        card.style.transform = RECEDE[i];
        card.style.zIndex = Z_RECEDE[i];
        const bw = card.querySelector(".benefits-wrapper");
        if (bw) { bw.style.maxHeight = "0"; bw.style.opacity = "0"; }
        card.querySelectorAll(".benefit-item")
            .forEach(el => el.classList.remove("revealed"));
      }
    });
  }

  cards.forEach((card, i) => {
    card.addEventListener("mouseenter", () => applyHover(i));
    card.addEventListener("mouseleave", applyResting);
    card.addEventListener("click", () => applyHover(i));
  });

  // Magnetic CTA buttons
  if (window.gsap && !window.matchMedia("(pointer: coarse)").matches) {
    document.querySelectorAll("[data-magnetic]").forEach(el => {
      const xTo = gsap.quickTo(el, "x", { duration: 0.9, ease: "elastic.out(1, 0.3)" });
      const yTo = gsap.quickTo(el, "y", { duration: 0.9, ease: "elastic.out(1, 0.3)" });
      el.addEventListener("mousemove", e => {
        const r = el.getBoundingClientRect();
        xTo(((e.clientX - r.left - r.width/2) / r.width) * 15);
        yTo(((e.clientY - r.top - r.height/2) / r.height) * 15);
      });
      el.addEventListener("mouseleave", () => { xTo(0); yTo(0); });
    });
  }
}

document.addEventListener("DOMContentLoaded", initPricingDeck);
