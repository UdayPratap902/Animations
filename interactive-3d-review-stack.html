3D Perspective Review Stack Architecture & React Guide

This animation architecture combines CSS 3D perspective (perspective: 1200px), 3-tier depth coordinates, dynamic cursor-tracked radial glare, spring transitions, and drag/touch gestures.

01. Mathematical 3D Variant Matrix

Position Relative to Active X Translation Scale rotateY Tilt Z-Index Opacity / Blur
Center (Active, diff = 0) 0% 1.00 0 deg (+ mouse tilt) 10 1.0 / 0px
Left Wing (diff = -1 or N-1) -52% 0.88 +10 deg 5 0.65 / 1.5px blur
Right Wing (diff = +1) +52% 0.88 -10 deg 5 0.65 / 1.5px blur
Background Queue ± 85% 0.75 0 deg 1 0.0 / 4px blur

02. Production Component Code

GlobalReviewsStack.tsx (From AvadaKeDevara Frontend) 
import React, { useState, useEffect, useRef } from "react";
import {
  motion,
  AnimatePresence,
  PanInfo,
  useMotionValue,
  useTransform,
} from "framer-motion";
import { Star, ChevronLeft, ChevronRight, Quote, CheckCircle2, Sparkles } from "lucide-react";

interface Testimonial {
  id: string;
  name: string;
  role: string;
  category: string;
  highlight?: string;
  text: string;
  initials: string;
  rating: number;
}

export const GlobalReviewsStack: React.FC<{ testimonials: Testimonial[] }> = ({ testimonials }) => {
  const [currentIndex, setCurrentIndex] = useState<number>(0);
  const [isHovered, setIsHovered] = useState<boolean>(false);
  const [direction, setDirection] = useState<number>(0);
  const total = testimonials.length;
  const autoPlayRef = useRef<number | undefined>(undefined);

  useEffect(() => {
    if (isHovered || total <= 1) {
      if (autoPlayRef.current) clearInterval(autoPlayRef.current);
      return;
    }
    autoPlayRef.current = window.setInterval(() => {
      setDirection(1);
      setCurrentIndex((prev) => (prev + 1) % total);
    }, 4500);
    return () => {
      if (autoPlayRef.current) clearInterval(autoPlayRef.current);
    };
  }, [isHovered, total]);

  const handleNext = () => {
    setDirection(1);
    setCurrentIndex((prev) => (prev + 1) % total);
  };

  const handlePrev = () => {
    setDirection(-1);
    setCurrentIndex((prev) => (prev - 1 + total) % total);
  };

  const handleDragEnd = (_e: any, { offset }: PanInfo) => {
    if (offset.x < -50) handleNext();
    else if (offset.x > 50) handlePrev();
  };

  const getVariants = (index: number) => {
    const diff = (index - currentIndex + total) % total;
    if (diff === 0) {
      return { x: "0%", scale: 1, zIndex: 10, opacity: 1, filter: "blur(0px)", rotateY: 0 };
    } else if (diff === total - 1 || diff === -1) {
      return { x: "-52%", scale: 0.88, zIndex: 5, opacity: 0.6, filter: "blur(2px)", rotateY: 10 };
    } else if (diff === 1) {
      return { x: "52%", scale: 0.88, zIndex: 5, opacity: 0.6, filter: "blur(2px)", rotateY: -10 };
    } else {
      return { x: diff > 1 && diff < total / 2 ? "85%" : "-85%", scale: 0.75, zIndex: 1, opacity: 0, filter: "blur(4px)", rotateY: 0 };
    }
  };

  // 3D Tilt & Radial Glare Motion Values
  const x = useMotionValue(0);
  const y = useMotionValue(0);
  const rotateX = useTransform(y, [-100, 100], [5, -5]);
  const rotateY = useTransform(x, [-100, 100], [-5, 5]);
  const glareX = useTransform(x, [-100, 100], [100, -100]);
  const glareY = useTransform(y, [-100, 100], [100, -100]);

  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
    const rect = e.currentTarget.getBoundingClientRect();
    x.set(e.clientX - (rect.left + rect.width / 2));
    y.set(e.clientY - (rect.top + rect.height / 2));
  };

  const handleMouseLeave = () => {
    x.set(0);
    y.set(0);
  };

  return (
    <div
      className="relative w-full h-[400px] flex items-center justify-center select-none"
      style={{ perspective: 1200 }}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      <div className="relative w-full max-w-5xl h-full flex items-center justify-center">
        <AnimatePresence initial={false} custom={direction}>
          {testimonials.map((t, idx) => {
            const diff = (idx - currentIndex + total) % total;
            const isCenter = diff === 0;

            return (
              <motion.div
                key={t.id}
                className="absolute w-[88%] sm:w-[76%] md:w-[62%] max-w-xl rounded-3xl p-6 md:p-8 shadow-2xl border border-border/90 bg-white/95 backdrop-blur-xl flex flex-col justify-between overflow-hidden cursor-pointer"
                animate={getVariants(idx)}
                style={isCenter ? { rotateX, rotateY } : {}}
                transition={{ type: "spring", stiffness: 300, damping: 30 }}
                drag={isCenter ? "x" : false}
                dragConstraints={{ left: 0, right: 0 }}
                onDragEnd={handleDragEnd}
                onClick={() => {
                  if (diff === 1) handleNext();
                  if (diff === total - 1) handlePrev();
                }}
                onMouseMove={isCenter ? handleMouseMove : undefined}
                onMouseLeave={isCenter ? handleMouseLeave : undefined}
              >
                {isCenter && (
                  <motion.div
                    className="absolute inset-0 z-20 pointer-events-none opacity-0 hover:opacity-100 transition-opacity rounded-3xl"
                    style={{
                      background: "radial-gradient(circle at center, rgba(59,130,246,0.15) 0%, rgba(255,255,255,0) 65%)",
                      x: glareX,
                      y: glareY,
                    }}
                  />
                )}

                <div className="flex items-center justify-between pb-3 border-b border-border/50">
                  <span className="text-xs font-bold uppercase text-primary">{t.category}</span>
                  <span className="text-xs font-mono text-slate-400">0{idx + 1}/0{total}</span>
                </div>

                <p className="font-serif italic text-sm md:text-base text-slate-800 my-4">"{t.text}"</p>

                <div className="flex items-center justify-between pt-3 border-t border-border/50">
                  <div className="flex items-center gap-3">
                    <div className="w-10 h-10 rounded-full bg-primary/20 text-primary font-bold flex items-center justify-center">{t.initials}</div>
                    <div>
                      <h4 className="font-bold text-sm text-slate-900">{t.name}</h4>
                      <p className="text-xs text-slate-500">{t.role}</p>
                    </div>
                  </div>
                </div>
              </motion.div>
            );
          })}
        </AnimatePresence>
      </div>

      <button onClick={handlePrev} className="absolute left-2 w-12 h-12 rounded-full bg-white border shadow-lg flex items-center justify-center z-40">
        <ChevronLeft className="w-6 h-6" />
      </button>
      <button onClick={handleNext} className="absolute right-2 w-12 h-12 rounded-full bg-white border shadow-lg flex items-center justify-center z-40">
        <ChevronRight className="w-6 h-6" />
      </button>
    </div>
  );
};

// <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.5/gsap.min.js"></script>

let activeIdx = 0;
const totalCards = 4;

function renderCarousel(animate = true) {
  for (let i = 0; i < totalCards; i++) {
    const card = document.getElementById(`card-${i}`);
    const dot = document.getElementById(`dot-${i}`);
    const diff = (i - activeIdx + totalCards) % totalCards;

    let target = {};
    if (diff === 0) {
      // Center Active Card
      target = {
        xPercent: 0,
        scale: 1,
        rotateY: 0,
        rotateX: 0,
        opacity: 1,
        zIndex: 10,
        filter: "blur(0px) brightness(1)",
        duration: animate ? 0.7 : 0,
        ease: "power3.out",
        overwrite: "auto",
      };
      if (dot) dot.className = "h-2 rounded-full transition-all duration-300 w-8 bg-ink shadow-2xs";
    } else if (diff === totalCards - 1 || diff === -1) {
      // Left Wing Card
      target = {
        xPercent: -52,
        scale: 0.88,
        rotateY: 12,
        rotateX: 0,
        opacity: 0.65,
        zIndex: 5,
        filter: "blur(1.5px) brightness(0.98)",
        duration: animate ? 0.7 : 0,
        ease: "power3.out",
        overwrite: "auto",
      };
      if (dot) dot.className = "h-2 rounded-full transition-all duration-300 w-2 bg-warm-300 hover:bg-warm-400";
    } else if (diff === 1) {
      // Right Wing Card
      target = {
        xPercent: 52,
        scale: 0.88,
        rotateY: -12,
        rotateX: 0,
        opacity: 0.65,
        zIndex: 5,
        filter: "blur(1.5px) brightness(0.98)",
        duration: animate ? 0.7 : 0,
        ease: "power3.out",
        overwrite: "auto",
      };
      if (dot) dot.className = "h-2 rounded-full transition-all duration-300 w-2 bg-warm-300 hover:bg-warm-400";
    } else {
      // Background Queue
      const xP = (diff > 1 && diff < totalCards / 2) ? 85 : -85;
      target = {
        xPercent: xP,
        scale: 0.75,
        rotateY: 0,
        rotateX: 0,
        opacity: 0,
        zIndex: 1,
        filter: "blur(4px) brightness(0.9)",
        duration: animate ? 0.6 : 0,
        ease: "power3.inOut",
        overwrite: "auto",
      };
      if (dot) dot.className = "h-2 rounded-full transition-all duration-300 w-2 bg-warm-300 hover:bg-warm-400";
    }

    gsap.to(card, target);
  }
}
